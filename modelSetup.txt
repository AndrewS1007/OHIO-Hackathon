# 1. Install Ollama (Linux/WSL)
curl -fsSL https://ollama.com/install.sh | sh

# 2. Download the Qwen 2.5 3B model and build our chatbot model from it
ollama pull qwen2.5:3b
ollama create outage-concierge -f Modelfile
# (Each command prints "success" when done. The Ollama server keeps running in the background.)

# 3. Activate the virtual environment (if not already active)
source .venv/bin/activate

# 4. Start the backend with the AI turned on
CONCIERGE_USE_LLM=1 python -m uvicorn concierge.api:app --reload
